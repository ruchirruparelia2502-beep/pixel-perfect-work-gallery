{

  (function initGovernanceWorkflow() {
    // 1. Toast Notification Helper
    function showToast(title, message, isSuccess = true) {
      const container = document.getElementById('toastContainer');
      if (!container) return;
      
      const toast = document.createElement('div');
      toast.className = `p-3.5 rounded-2xl shadow-xl flex items-start gap-2.5 text-xs transition transform duration-300 pointer-events-auto max-w-sm ${
        isSuccess ? 'bg-[#18181A] text-white' : 'bg-[#C64977] text-white'
      }`;
      toast.innerHTML = `
        <span class="material-symbols-outlined text-base ${isSuccess ? 'text-[#F5A3BE]' : 'text-white'}">
          ${isSuccess ? 'check_circle' : 'warning'}
        </span>
        <div class="flex-1">
          <p class="font-bold">${title}</p>
          <p class="text-[11px] opacity-90 mt-0.5">${message}</p>
        </div>
      `;
      container.appendChild(toast);
      
      setTimeout(() => {
        toast.style.opacity = '0';
        toast.style.transform = 'translateY(10px)';
        setTimeout(() => toast.remove(), 300);
      }, 4000);
    }

    // 2. Character counter & validation
    const commentBox = document.getElementById('governanceComment');
    const charCount = document.getElementById('charCount');
    const errorMsg = document.getElementById('commentErrorMsg');

    if (commentBox && charCount) {
      commentBox.addEventListener('input', () => {
        const len = commentBox.value.trim().length;
        charCount.textContent = `${len} chars`;
        if (len >= 15 && errorMsg) {
          errorMsg.classList.add('hidden');
        }
      });
    }

    // 3. Role Switcher Logic
    let currentRole = 'reviewer'; // default active
    const roleButtons = document.querySelectorAll('.role-btn');
    const personaBadge = document.getElementById('activePersonaBadge');
    const roleNoticeTag = document.getElementById('roleNoticeTag');
    const roleActionFeedback = document.getElementById('roleActionFeedback');

    const btnForwardHod = document.getElementById('btnForwardHod');
    const btnReturnComment = document.getElementById('btnReturnComment');
    const btnAuthorSubmit = document.getElementById('btnAuthorSubmit');
    const btnHodApprove = document.getElementById('btnHodApprove');

    const wrapAuthorSubmit = document.getElementById('wrapAuthorSubmit');
    const wrapHodApprove = document.getElementById('wrapHodApprove');

    function updateRoleUI(role) {
      currentRole = role;

      // Update switcher pills
      roleButtons.forEach(btn => {
        const btnRole = btn.getAttribute('data-role');
        if (btnRole === role) {
          btn.className = 'role-btn px-3 py-1 rounded-xl text-xs font-bold bg-[#18181A] text-[#F5A3BE] shadow-xs transition cursor-pointer';
          btn.innerHTML = btn.textContent.replace(' ACTIVE', '') + ' <span class="text-[9px] bg-[#F5A3BE]/20 px-1 py-0.2 rounded-md ml-1 font-extrabold">ACTIVE</span>';
        } else {
          btn.className = 'role-btn px-3 py-1 rounded-xl text-xs font-semibold text-zinc-600 hover:text-zinc-950 transition cursor-pointer';
          btn.innerHTML = btn.textContent.replace(' ACTIVE', '');
        }
      });

      // Update persona badge and actions based on role
      if (role === 'reviewer') {
        personaBadge.innerHTML = `<span class="w-2 h-2 rounded-full bg-[#F5A3BE]"></span><span>Dr. Aris Thorne (Reviewer)</span>`;
        roleNoticeTag.textContent = 'Reviewer Stage Controls';
        roleActionFeedback.innerHTML = `<span class="material-symbols-outlined text-sm text-[#A62758]">info</span><span>Reviewer permissions verified. Submitting endorsement will push course state to <strong class="text-zinc-900">Stage 4: Pending HOD</strong>.</span>`;
        
        // Reviewer controls
        enableButton(btnForwardHod, true, 'Forward to HOD (Endorse)');
        enableButton(btnReturnComment, true, 'Return with Comment');
        disableButton(btnAuthorSubmit, 'Disabled: Only Course Author (Prof. Marcus Vance) can resubmit draft');
        disableButton(btnHodApprove, 'Disabled: Requires HOD (Dr. Eleanor Sterling) credentials once stage reaches Pending HOD');
      } 
      else if (role === 'author') {
        personaBadge.innerHTML = `<span class="w-2 h-2 rounded-full bg-[#2F6497]"></span><span>Prof. Marcus Vance (Author)</span>`;
        roleNoticeTag.textContent = 'Course Author Controls';
        roleActionFeedback.innerHTML = `<span class="material-symbols-outlined text-sm text-[#2F6497]">info</span><span>You are viewing draft author tools. Resubmitting will trigger re-inspection of Bloom taxonomy.</span>`;
        
        disableButton(btnForwardHod, 'Disabled: Only assigned reviewer can forward to HOD');
        disableButton(btnReturnComment, 'Disabled: Author cannot return their own syllabus');
        enableButton(btnAuthorSubmit, false, 'Submit for Review');
        disableButton(btnHodApprove, 'Disabled: Author cannot ratify syllabus');
      } 
      else if (role === 'hod') {
        personaBadge.innerHTML = `<span class="w-2 h-2 rounded-full bg-[#8C6906]"></span><span>Dr. Eleanor Sterling (Head of Dept)</span>`;
        roleNoticeTag.textContent = 'Department Head Approval Matrix';
        roleActionFeedback.innerHTML = `<span class="material-symbols-outlined text-sm text-[#8C6906]">info</span><span>HOD credentials detected. Ratification confers Senate registration clearance.</span>`;
        
        disableButton(btnForwardHod, 'Disabled: Reviewer has already signed off or stage passed');
        enableButton(btnReturnComment, true, 'Return to Author (Remediation)');
        disableButton(btnAuthorSubmit, 'Disabled: Only Course Author can resubmit draft');
        enableButton(btnHodApprove, false, 'Approve & Ratify (Senate Seal)');
      }
    }

    function enableButton(btn, isDark, labelText) {
      if (!btn) return;
      btn.disabled = false;
      btn.classList.remove('bg-zinc-200', 'text-zinc-400', 'cursor-not-allowed');
      btn.classList.add('cursor-pointer');
      
      if (isDark) {
        btn.classList.add('bg-[#18181A]', 'text-[#F5A3BE]', 'hover:bg-black');
      } else {
        btn.classList.add('bg-zinc-900', 'text-white', 'hover:bg-zinc-800');
      }
      
      const span = btn.querySelector('span:not(.material-symbols-outlined)');
      if (span && labelText) span.textContent = labelText;

      const parentWrap = btn.closest('.group');
      if (parentWrap) {
        const tooltip = parentWrap.querySelector('.tooltip');
        if (tooltip) tooltip.classList.add('hidden');
      }
    }

    function disableButton(btn, tooltipText) {
      if (!btn) return;
      btn.disabled = true;
      btn.className = 'action-btn px-4 py-2.5 rounded-full bg-zinc-200 text-zinc-400 text-xs font-bold cursor-not-allowed flex items-center gap-2 transition';
      
      const parentWrap = btn.closest('.group');
      if (parentWrap) {
        const tooltip = parentWrap.querySelector('.tooltip');
        if (tooltip) {
          tooltip.classList.remove('hidden');
          if (tooltipText) tooltip.textContent = tooltipText;
        }
      }
    }

    roleButtons.forEach(btn => {
      btn.addEventListener('click', () => {
        const selected = btn.getAttribute('data-role');
        updateRoleUI(selected);
        showToast('Persona Switched', `Simulating view as ${btn.textContent.trim().replace('ACTIVE', '')}`, true);
      });
    });

    // 4. Return Loop Simulation Toggle
    let isReturnedState = false;
    const toggleReturnedBtn = document.getElementById('toggleReturnedStateBtn');
    const returnedToggleLabel = document.getElementById('returnedToggleLabel');
    const alertBadge = document.getElementById('remediationAlertBadge');
    const stepReview = document.getElementById('stepUnderReview');

    if (toggleReturnedBtn) {
      toggleReturnedBtn.addEventListener('click', () => {
        isReturnedState = !isReturnedState;
        if (isReturnedState) {
          alertBadge.classList.remove('hidden');
          returnedToggleLabel.textContent = 'Exit Returned Loop';
          toggleReturnedBtn.classList.remove('bg-[#FCE8E6]', 'text-[#A62758]');
          toggleReturnedBtn.classList.add('bg-[#C64977]', 'text-white');
          
          if (stepReview) {
            stepReview.classList.remove('bg-[#FDECEF]');
            stepReview.classList.add('bg-[#FCE8E6]', 'ring-2', 'ring-[#C64977]');
          }
          showToast('Course Flagged', 'Status set to: Returned (Remediation Required to Author)', false);
        } else {
          alertBadge.classList.add('hidden');
          returnedToggleLabel.textContent = 'Simulate Returned Loop';
          toggleReturnedBtn.classList.add('bg-[#FCE8E6]', 'text-[#A62758]');
          toggleReturnedBtn.classList.remove('bg-[#C64977]', 'text-white');
          
          if (stepReview) {
            stepReview.classList.add('bg-[#FDECEF]');
            stepReview.classList.remove('bg-[#FCE8E6]', 'ring-2', 'ring-[#C64977]');
          }
          showToast('Loop Cleared', 'Status returned to: Under Review (Peer Reviewer)', true);
        }
      });
    }

    // 5. Action Button Handlers
    if (btnForwardHod) {
      btnForwardHod.addEventListener('click', () => {
        const val = commentBox ? commentBox.value.trim() : '';
        if (val.length < 15) {
          if (errorMsg) errorMsg.classList.remove('hidden');
          if (commentBox) commentBox.focus();
          showToast('Endorsement Incomplete', 'Please provide a valid reviewer note before forwarding.', false);
          return;
        }
        showToast('Forwarded to HOD', 'Dr. Aris Thorne formally endorsed syllabus. Stage progressed to Pending HOD.', true);
      });
    }

    if (btnReturnComment) {
      btnReturnComment.addEventListener('click', () => {
        const val = commentBox ? commentBox.value.trim() : '';
        if (val.length < 15) {
          if (errorMsg) errorMsg.classList.remove('hidden');
          if (commentBox) commentBox.focus();
          showToast('Comment Required', 'A justification note is strictly mandatory to return the course.', false);
          return;
        }
        showToast('Returned to Author', 'Course specification dispatched to Prof. Vance for remediation.', false);
      });
    }

    if (btnAuthorSubmit) {
      btnAuthorSubmit.addEventListener('click', () => {
        if (!btnAuthorSubmit.disabled) {
          showToast('Submission Dispatched', 'Course revision submitted to Peer Review pool.', true);
        }
      });
    }

    if (btnHodApprove) {
      btnHodApprove.addEventListener('click', () => {
        if (!btnHodApprove.disabled) {
          showToast('Senate Ratified!', 'HOD sign-off applied with cryptographic signature.', true);
        }
      });
    }

    // 6. Timeline Filtering
    const filterChips = document.querySelectorAll('.filter-chip');
    const timelineItems = document.querySelectorAll('.timeline-item');

    filterChips.forEach(chip => {
      chip.addEventListener('click', () => {
        const cat = chip.getAttribute('data-filter');
        filterChips.forEach(c => {
          c.className = 'filter-chip px-2.5 py-1 rounded-full text-[11px] font-semibold bg-white text-zinc-600 hover:text-zinc-900 shadow-2xs transition cursor-pointer';
        });
        chip.className = 'filter-chip px-2.5 py-1 rounded-full text-[11px] font-bold bg-[#18181A] text-white shadow-2xs transition cursor-pointer';

        timelineItems.forEach(item => {
          const itemCat = item.getAttribute('data-category');
          if (cat === 'all' || itemCat === cat) {
            item.style.display = 'block';
          } else {
            item.style.display = 'none';
          }
        });
      });
    });

    // 7. General button feedback toasts
    const auditBtn = document.getElementById('auditBtn');
    if (auditBtn) {
      auditBtn.addEventListener('click', () => {
        showToast('OBE Engine Executed', 'Taxonomy verified: 0 critical defects. Score 92%.', true);
      });
    }

    const exportBtn = document.getElementById('exportDossierBtn');
    if (exportBtn) {
      exportBtn.addEventListener('click', () => {
        showToast('Compiling Dossier', 'CS304 Academic Senate Dossier PDF generated successfully.', true);
      });
    }

    const viewLogsBtn = document.getElementById('viewCompleteLogsBtn');
    if (viewLogsBtn) {
      viewLogsBtn.addEventListener('click', () => {
        showToast('Export Complete', 'Audit ledger exported as CS304_audit_ledger.json', true);
      });
    }

    // Initialize with reviewer role
    updateRoleUI('reviewer');
  })();

}
{

  // State flag to track whether user manually toggled calendar
  let userManuallyToggled = false;
  let isCalendarContracted = false;

  function updateCalendarButtonState(contracted) {
    const toggleText = document.getElementById('toggleCalendarText');
    const toggleBtn = document.getElementById('hide-calendar-btn');
    const toggleIcon = document.getElementById('toggleCalendarIcon');

    if (contracted) {
      if (toggleText) toggleText.textContent = 'Show Calendar';
      if (toggleIcon) toggleIcon.textContent = 'calendar_month';
      if (toggleBtn) {
        toggleBtn.setAttribute('title', 'Expand Calendar Panel');
        toggleBtn.classList.remove('bg-white', 'text-zinc-800');
        toggleBtn.classList.add('bg-zinc-900', 'text-white', 'ring-2', 'ring-[#F5A3BE]/40');
      }
    } else {
      if (toggleText) toggleText.textContent = 'Hide Calendar';
      if (toggleIcon) toggleIcon.textContent = 'view_sidebar';
      if (toggleBtn) {
        toggleBtn.setAttribute('title', 'Hide Calendar Panel');
        toggleBtn.classList.remove('bg-zinc-900', 'text-white', 'ring-2', 'ring-[#F5A3BE]/40');
        toggleBtn.classList.add('bg-white', 'text-zinc-800');
      }
    }
  }

  // Toggle Calendar Functionality
  function toggleCalendar(forceState) {
    const sidebar = document.getElementById('calendar-sidebar');
    if (!sidebar) return;

    if (typeof forceState === 'boolean') {
      isCalendarContracted = forceState;
    } else {
      userManuallyToggled = true; // explicitly clicked by user
      isCalendarContracted = !isCalendarContracted;
    }

    if (isCalendarContracted) {
      sidebar.classList.add('contracted-sidebar');
    } else {
      sidebar.classList.remove('contracted-sidebar');
    }

    updateCalendarButtonState(isCalendarContracted);
  }

  // Expose toggleCalendar globally
  window.toggleCalendar = toggleCalendar;
  window.toggleCalendarSidebar = toggleCalendar;

  // Auto-contraction on scroll logic
  function setupScrollAutoContract() {
    const scrollContainer = document.getElementById('mainWorkspace') || window;
    const scrollThreshold = 140; // threshold past top hero/header

    const handleScroll = () => {
      // Determine vertical scroll distance
      const scrollTop = scrollContainer === window 
        ? (window.pageYOffset || document.documentElement.scrollTop) 
        : scrollContainer.scrollTop;

      // If user hasn't explicitly locked state by manual click, or when they return to top
      if (scrollTop > scrollThreshold) {
        if (!isCalendarContracted && !userManuallyToggled) {
          toggleCalendar(true); // auto contract smoothly
        }
      } else if (scrollTop <= 40) {
        // When scrolled back to the very top, auto-expand back and reset manual override lock
        if (isCalendarContracted) {
          userManuallyToggled = false;
          toggleCalendar(false); // auto restore
        }
      }
    };

    if (scrollContainer === window) {
      window.addEventListener('scroll', handleScroll, { passive: true });
    } else {
      scrollContainer.addEventListener('scroll', handleScroll, { passive: true });
      window.addEventListener('scroll', handleScroll, { passive: true });
    }
  }

  __sfReady( () => {
    setupScrollAutoContract();

    // Bento card expansion handling if present
    const cards = document.querySelectorAll('.bento-card');
    let activeCardId = null;

    const initialExpanded = document.querySelector('.bento-card.is-expanded');
    if (initialExpanded) {
      activeCardId = initialExpanded.getAttribute('data-card-id');
    }

    function resetCards() {
      activeCardId = null;
      cards.forEach(card => {
        card.classList.remove('is-expanded', 'is-dimmed', 'ring-2', 'ring-zinc-900/30', 'shadow-xl');
        const defaultCol = card.getAttribute('data-default-col');
        if (defaultCol) {
          card.className = card.className.replace(/lg:col-span-\d+/g, '').trim();
          card.classList.add(defaultCol);
        }
        const badge = card.querySelector('.card-badge');
        if (badge) badge.classList.add('hidden');
        const path = card.querySelector('.expand-icon-path');
        if (path) {
          path.setAttribute('d', 'M15 3h6v6M9 21H3v-6M21 3l-7 7M3 21l7-7');
        }
      });
    }

    function expandCard(card) {
      const cardId = card.getAttribute('data-card-id');
      if (activeCardId === cardId) {
        resetCards();
        return;
      }
      activeCardId = cardId;
      cards.forEach(c => {
        const cId = c.getAttribute('data-card-id');
        const defaultCol = c.getAttribute('data-default-col');
        if (defaultCol) {
          c.className = c.className.replace(/lg:col-span-\d+/g, '').trim();
        }
        const badge = c.querySelector('.card-badge');
        const path = c.querySelector('.expand-icon-path');

        if (cId === cardId) {
          c.classList.remove('is-dimmed');
          c.classList.add('is-expanded', 'ring-2', 'ring-zinc-900/30', 'shadow-xl');
          c.classList.add('lg:col-span-12');
          if (badge) badge.classList.remove('hidden');
          if (path) path.setAttribute('d', 'M18 6L6 18M6 6l12 12');
        } else {
          c.classList.remove('is-expanded', 'ring-2', 'ring-zinc-900/30', 'shadow-xl');
          c.classList.add('is-dimmed');
          c.classList.add('lg:col-span-4');
          if (badge) badge.classList.add('hidden');
          if (path) path.setAttribute('d', 'M15 3h6v6M9 21H3v-6M21 3l-7 7M3 21l7-7');
        }
      });
    }

    cards.forEach(card => {
      card.addEventListener('click', (e) => {
        if (e.target.closest('button:not(.toggle-btn)') || e.target.closest('a')) return;
        expandCard(card);
      });
    });

    document.addEventListener('click', (e) => {
      if (!e.target.closest('#bentoGrid')) {
        if (activeCardId !== null) resetCards();
      }
    });

    document.addEventListener('keydown', (e) => {
      if (e.key === 'Escape' && activeCardId !== null) resetCards();
    });

    // Modal Handlers
    const aiModal = document.getElementById('aiSettingsModal');
    const aiTrigger = document.getElementById('aiSettingsTriggerBtn');
    const aiClose = document.getElementById('closeSettingsBtn');
    const saveSettingsBtn = document.getElementById('saveSettingsBtn');

    if (aiTrigger && aiModal) {
      aiTrigger.addEventListener('click', () => {
        aiModal.classList.remove('hidden');
        aiModal.classList.add('flex');
      });
    }
    if (aiClose && aiModal) {
      aiClose.addEventListener('click', () => {
        aiModal.classList.add('hidden');
        aiModal.classList.remove('flex');
      });
    }
    if (saveSettingsBtn && aiModal) {
      saveSettingsBtn.addEventListener('click', () => {
        aiModal.classList.add('hidden');
        aiModal.classList.remove('flex');
      });
    }

    const eventModal = document.getElementById('addEventModal');
    const closeEventBtn = document.getElementById('closeAddEventBtn');
    const cancelEventBtn = document.getElementById('cancelAddEventBtn');
    const confirmAddEventBtn = document.getElementById('confirmAddEventBtn');

    [closeEventBtn, cancelEventBtn].forEach(btn => {
      if (btn && eventModal) {
        btn.addEventListener('click', () => {
          eventModal.classList.add('hidden');
          eventModal.classList.remove('flex');
        });
      }
    });

    if (confirmAddEventBtn && eventModal) {
      confirmAddEventBtn.addEventListener('click', () => {
        eventModal.classList.add('hidden');
        eventModal.classList.remove('flex');
        const toast = document.getElementById('toastContainer');
        if (toast) {
          const item = document.createElement('div');
          item.className = 'p-3 rounded-2xl bg-[#18181A] text-white text-xs shadow-xl flex items-center gap-2';
          item.innerHTML = '<span class="material-symbols-outlined text-[#F5A3BE] text-base">event_available</span><span>Milestone registered on calendar!</span>';
          toast.appendChild(item);
          setTimeout(() => item.remove(), 3500);
        }
      });
    }
  });
