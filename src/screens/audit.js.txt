  // Toast Helper
  function showToast(message, type = 'success') {
    const container = document.getElementById('toastContainer');
    if (!container) return;

    const toast = document.createElement('div');
    toast.className = `px-4 py-2.5 rounded-2xl text-xs font-bold shadow-lg pointer-events-auto flex items-center gap-2 transition-all transform duration-300 translate-y-4 opacity-0 ${
      type === 'success' ? 'bg-[#18181A] text-white' : 'bg-[#C64977] text-white'
    }`;
    toast.innerHTML = `<span class="material-symbols-outlined text-sm">${type === 'success' ? 'check_circle' : 'info'}</span> ${message}`;

    container.appendChild(toast);
    setTimeout(() => {
      toast.classList.remove('translate-y-4', 'opacity-0');
    }, 10);

    setTimeout(() => {
      toast.classList.add('translate-y-4', 'opacity-0');
      setTimeout(() => toast.remove(), 300);
    }, 3000);
  }

  // Active state filters
  let currentSeverity = 'all';

  function setSeverityFilter(sev) {
    currentSeverity = sev;
    document.querySelectorAll('.sev-btn').forEach(btn => {
      if (btn.getAttribute('data-severity') === sev) {
        btn.classList.add('bg-zinc-900', 'text-white');
        btn.classList.remove('bg-white', 'text-zinc-700');
      } else {
        btn.classList.remove('bg-zinc-900', 'text-white');
        btn.classList.add('bg-white', 'text-zinc-700');
      }
    });
    applyFilters();
  }

  function filterBySeverityDirect(sev) {
    setSeverityFilter(sev);
  }

  function applyFilters() {
    const searchTerm = (document.getElementById('defectSearchInput')?.value || '').toLowerCase().trim();
    const selectedSection = document.getElementById('sectionFilterSelect')?.value || 'all';
    const aiOnly = document.getElementById('aiRemediatedToggle')?.checked || false;

    const defectRows = document.querySelectorAll('.defect-row');
    const sectionGroups = document.querySelectorAll('.section-group');
    const passedContainer = document.getElementById('passedChecksContainer');

    if (currentSeverity === 'passed') {
      // Hide all regular defect section groups, expand passed checks
      sectionGroups.forEach(g => g.style.display = 'none');
      if (passedContainer) {
        passedContainer.style.display = 'block';
        const content = document.getElementById('passedChecksContent');
        if (content && content.classList.contains('hidden')) {
          togglePassedAccordion();
        }
      }
      return;
    } else {
      if (passedContainer) passedContainer.style.display = 'block';
    }

    defectRows.forEach(row => {
      const rowSev = row.getAttribute('data-severity');
      const rowAi = row.getAttribute('data-ai') === 'true';
      const rowText = (row.getAttribute('data-text') || '').toLowerCase();

      const matchesSev = currentSeverity === 'all' || rowSev === currentSeverity;
      const matchesSearch = !searchTerm || rowText.includes(searchTerm);
      const matchesAi = !aiOnly || rowAi;

      if (matchesSev && matchesSearch && matchesAi) {
        row.style.display = '';
      } else {
        row.style.display = 'none';
      }
    });

    // Handle Section groups visibility
    sectionGroups.forEach(group => {
      const sectionId = group.getAttribute('data-section-id');
      const matchesSection = selectedSection === 'all' || selectedSection === sectionId;

      const visibleChildRows = group.querySelectorAll('.defect-row:not([style*="display: none"])');
      if (matchesSection && visibleChildRows.length > 0) {
        group.style.display = '';
      } else {
        group.style.display = 'none';
      }
    });
  }

  // Toggle Accordion for Passed Checks
  function togglePassedAccordion() {
    const content = document.getElementById('passedChecksContent');
    const chevron = document.getElementById('accordionChevron');
    if (!content) return;

    if (content.classList.contains('hidden')) {
      content.classList.remove('hidden');
      if (chevron) chevron.style.transform = 'rotate(180deg)';
    } else {
      content.classList.add('hidden');
      if (chevron) chevron.style.transform = 'rotate(0deg)';
    }
  }

  // Retractable calendar panel integration with app-shell
  function toggleCalendarPanel(forceState) {
    const sidebar = document.getElementById('calendar-sidebar');
    const label = document.getElementById('calendarToggleLabel');
    if (!sidebar) return;

    const isContracted = sidebar.classList.contains('contracted');
    const shouldContract = forceState !== undefined ? forceState : !isContracted;

    if (shouldContract) {
      sidebar.classList.add('contracted');
      if (label) label.textContent = 'Show Calendar';
    } else {
      sidebar.classList.remove('contracted');
      if (label) label.textContent = 'Hide Calendar';
    }
  }

  function triggerAuditRerun() {
    showToast('Running OBE audit across 5 course units...');

  }

  function exportAuditReport() {

  }

  function acceptFix(btn, code) {
    const row = btn.closest('.defect-row');
    if (row) {
      row.style.opacity = '0.4';
      row.style.pointerEvents = 'none';
      btn.textContent = 'Accepted ✓';
      btn.classList.remove('bg-zinc-900', 'hover:bg-black');
      btn.classList.add('bg-emerald-700', 'text-white');
    }
    showToast(`Remediation accepted for ${code}`);
  }

  function revertFix(btn, code) {
    showToast(`Reverted fix for ${code}. Marked for manual curriculum review.`, 'warning');
  }

  function editRationale(code) {

  }

  function reviewSchedule(code) {

  function inspectDiff(code) {
    activeDiffCode = code;
    const modal = document.getElementById('diffInspectionModal');
    const data = diffData[code] || {
      code,
      title: 'Syllabus Diff Inspection',
      original: 'Draft curriculum text.',
      remediated: 'Remediated syllabus text.'
    };

    document.getElementById('diffModalCode').textContent = data.code;
    document.getElementById('diffModalTitle').textContent = data.title;
    document.getElementById('diffOriginalText').textContent = data.original;
    document.getElementById('diffRemediatedText').textContent = data.remediated;

    if (modal) {
      modal.classList.remove('hidden');
      modal.classList.add('flex');
    }
  }

  function closeDiffModal() {
    const modal = document.getElementById('diffInspectionModal');
    if (modal) {
      modal.classList.add('hidden');
      modal.classList.remove('flex');
    }
  }

  function confirmModalDiffAccept() {
    closeDiffModal();
    if (activeDiffCode) {
      showToast(`Remediation accepted for ${activeDiffCode}`);
    }
  }

  // Expose toggleCalendar globally
  window.toggleCalendar = function() {
    toggleCalendarPanel();
  };
  window.toggleCalendarSidebar = window.toggleCalendar;

  __sfReady( () => {
    // Auto-contraction on scroll logic
    const workspace = document.getElementById('mainWorkspace');
    let userManuallyToggled = false;

    // Record manual user toggle
    const toggleBtn = document.getElementById('toggleCalendarBtn');
    if (toggleBtn) {
      toggleBtn.addEventListener('click', () => {
        userManuallyToggled = true;
      });
    }

    if (workspace) {
      const SCROLL_THRESHOLD = 140; // Past breadcrumbs and top header

      workspace.addEventListener('scroll', () => {
        const scrollTop = workspace.scrollTop;
        const sidebar = document.getElementById('calendar-sidebar');
        if (!sidebar) return;

        // Auto contract smoothly when scrolling down past top section
        if (scrollTop > SCROLL_THRESHOLD) {
          if (!sidebar.classList.contains('contracted')) {
            toggleCalendarPanel(true);
          }
        } else {
          // Restore when scrolling back to the very top if user hasn't deliberately kept it closed
          if (sidebar.classList.contains('contracted') && !userManuallyToggled) {
            toggleCalendarPanel(false);
          }
        }
      }, { passive: true });
    }

    // AI Provider modal handlers
    const aiModal = document.getElementById('aiSettingsModal');
    const aiTrigger = document.getElementById('aiSettingsTriggerBtn');
    const aiClose = document.getElementById('closeSettingsBtn');
    const saveSettingsBtn = document.getElementById('saveSettingsBtn');

    if (aiTrigger && aiModal) {
      aiTrigger.addEventListener('click', () => {
        aiModal.classList.remove('hidden');
        aiModal.classList.add('flex');
      });
    }
    if (aiClose && aiModal) {
      aiClose.addEventListener('click', () => {
        aiModal.classList.add('hidden');
        aiModal.classList.remove('flex');
      });
    }
    if (saveSettingsBtn && aiModal) {
      saveSettingsBtn.addEventListener('click', () => {
        aiModal.classList.add('hidden');
        aiModal.classList.remove('flex');
      });
    }
  });
