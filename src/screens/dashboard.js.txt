{

  // State tracking for manual override and contraction
  let isCalendarContracted = false;
  let manuallyToggled = false;

  // Toggle Calendar Functionality
  function setCalendarState(contract) {
    const sidebar = document.getElementById('calendar-sidebar');
    const toggleText = document.getElementById('toggleCalendarText');
    const toggleBtn = document.getElementById('hide-calendar-btn');

    if (!sidebar) return;
    isCalendarContracted = contract;

    if (contract) {
      sidebar.classList.add('calendar-contracted');
      if (toggleText) toggleText.textContent = 'Show Calendar';
      if (toggleBtn) {
        toggleBtn.setAttribute('title', 'Show Calendar Panel');
        toggleBtn.classList.add('bg-zinc-900', 'text-white');
        toggleBtn.classList.remove('bg-white/80', 'text-zinc-800');
      }
    } else {
      sidebar.classList.remove('calendar-contracted');
      if (toggleText) toggleText.textContent = 'Hide Calendar';
      if (toggleBtn) {
        toggleBtn.setAttribute('title', 'Hide Calendar Panel');
        toggleBtn.classList.remove('bg-zinc-900', 'text-white');
        toggleBtn.classList.add('bg-white/80', 'text-zinc-800');
      }
    }
  }

  function toggleCalendar(forceClose = false) {
    if (forceClose) {
      manuallyToggled = true;
      setCalendarState(true);
      return;
    }
    manuallyToggled = true;
    setCalendarState(!isCalendarContracted);
  }

  // Expose toggleCalendar globally
  window.toggleCalendar = toggleCalendar;
  window.toggleCalendarSidebar = toggleCalendar;

  __sfReady( () => {
    // Scroll listener for auto-contraction past header
    const mainWorkspace = document.getElementById('mainWorkspace');
    const topHeader = document.getElementById('topHeaderBar');

    if (mainWorkspace) {
      mainWorkspace.addEventListener('scroll', () => {
        // Only auto-contract/expand if user has not explicitly locked or clicked
        const scrollThreshold = (topHeader ? topHeader.offsetHeight + 15 : 65);
        if (mainWorkspace.scrollTop > scrollThreshold) {
          if (!isCalendarContracted && !manuallyToggled) {
            setCalendarState(true);
          }
        } else {
          // Scrolled back near top
          if (isCalendarContracted && !manuallyToggled) {
            setCalendarState(false);
          }
          // Reset manual override once back at very top so auto-contract can re-engage seamlessly
          if (mainWorkspace.scrollTop <= 5) {
            manuallyToggled = false;
          }
        }
      });
    }

    const cards = document.querySelectorAll('.bento-card');
    let activeCardId = null;

    // Check if yellow card is initially expanded
    const initialExpanded = document.querySelector('.bento-card.is-expanded');
    if (initialExpanded) {
      activeCardId = initialExpanded.getAttribute('data-card-id');
    }

    function resetCards() {
      activeCardId = null;
      cards.forEach(card => {
        card.classList.remove('is-expanded', 'is-dimmed', 'ring-2', 'ring-zinc-900/30', 'shadow-xl');
        
        // Restore default column spans
        const defaultCol = card.getAttribute('data-default-col');
        card.className = card.className.replace(/lg:col-span-\d+/g, '').trim();
        card.classList.add(defaultCol);

        // Hide badges
        const badge = card.querySelector('.card-badge');
        if (badge) badge.classList.add('hidden');

        // Reset icon to expand icon
        const path = card.querySelector('.expand-icon-path');
        if (path) {
          path.setAttribute('d', 'M15 3h6v6M9 21H3v-6M21 3l-7 7M3 21l7-7');
        }
      });
    }

    function expandCard(card) {
      const cardId = card.getAttribute('data-card-id');

      // If already active, clicking again collapses it
      if (activeCardId === cardId) {
        resetCards();
        return;
      }

      activeCardId = cardId;

      cards.forEach(c => {
        const cId = c.getAttribute('data-card-id');
        const defaultCol = c.getAttribute('data-default-col');
        c.className = c.className.replace(/lg:col-span-\d+/g, '').trim();

        const badge = c.querySelector('.card-badge');
        const path = c.querySelector('.expand-icon-path');

        if (cId === cardId) {
          c.classList.remove('is-dimmed');
          c.classList.add('is-expanded', 'ring-2', 'ring-zinc-900/30', 'shadow-xl');
          c.classList.add('lg:col-span-12'); // Take full width prominence in grid
          if (badge) badge.classList.remove('hidden');
          if (path) {
            // Collapse cross icon '✕'
            path.setAttribute('d', 'M18 6L6 18M6 6l12 12');
          }
        } else {
          c.classList.remove('is-expanded', 'ring-2', 'ring-zinc-900/30', 'shadow-xl');
          c.classList.add('is-dimmed');
          // For inactive cards, distribute compactly
          c.classList.add('lg:col-span-4');
          if (badge) badge.classList.add('hidden');
          if (path) {
            path.setAttribute('d', 'M15 3h6v6M9 21H3v-6M21 3l-7 7M3 21l7-7');
          }
        }
      });
    }

    // Attach click listeners to cards and their toggle buttons
    cards.forEach(card => {
      card.addEventListener('click', (e) => {
        // Prevent click if clicking direct action links inside expanded area
        if (e.target.closest('button:not(.toggle-btn)') || e.target.closest('a')) {
          return;
        }
        expandCard(card);
      });
    });

    // Clicking outside the grid closes any open card
    document.addEventListener('click', (e) => {
      if (!e.target.closest('#bentoGrid')) {
        if (activeCardId !== null) {
          resetCards();
        }
      }
    });

    // Allow Esc key to contract cards
    document.addEventListener('keydown', (e) => {
      if (e.key === 'Escape' && activeCardId !== null) {
        resetCards();
      }
    });

    // AI Provider modal handlers
    const aiModal = document.getElementById('aiSettingsModal');
    const aiTrigger = document.getElementById('aiSettingsTriggerBtn');
    const aiClose = document.getElementById('closeSettingsBtn');
    const saveSettingsBtn = document.getElementById('saveSettingsBtn');

    if (aiTrigger && aiModal) {
      aiTrigger.addEventListener('click', () => {
        aiModal.classList.remove('hidden');
        aiModal.classList.add('flex');
      });
    }
    if (aiClose && aiModal) {
      aiClose.addEventListener('click', () => {
        aiModal.classList.add('hidden');
        aiModal.classList.remove('flex');
      });
    }
    if (saveSettingsBtn && aiModal) {
      saveSettingsBtn.addEventListener('click', () => {
        aiModal.classList.add('hidden');
        aiModal.classList.remove('flex');
      });
    }
  });

}
