  // Copy Direct Senate Link with Clipboard API
  function triggerSenateCopyLink() {
    const senateUrl = window.location.origin + window.location.pathname + '#senate-verify-cs304-r22';
    if (navigator.clipboard) {
      navigator.clipboard.writeText(senateUrl).then(() => {
        showToastNotification('Direct Senate Verification Link copied to clipboard!', 'link');
      }).catch(() => {
        showToastNotification('Senate link ready: ' + senateUrl, 'link');
      });
    } else {
      showToastNotification('Direct Senate Verification Link copied!', 'link');
    }
  }

  // Toggle Paper Contrast/Elevation
  function togglePaperElevation() {
    const sheet = document.getElementById('academicPaperSheet');
    if (!sheet) return;
    if (sheet.classList.contains('bg-[#FFFDF9]')) {
      sheet.classList.remove('bg-[#FFFDF9]');
      sheet.classList.add('bg-white', 'shadow-2xl');
      showToastNotification('High-contrast stark white mode active', 'contrast');
    } else {
      sheet.classList.remove('bg-white', 'shadow-2xl');
      sheet.classList.add('bg-[#FFFDF9]');
      showToastNotification('Warm editorial paper tone active', 'article');
    }
  }

  // Toggle Sheet Watermark
  function toggleSheetWatermark(isEnabled) {
    const sheet = document.getElementById('academicPaperSheet');
    if (!sheet) return;
    if (isEnabled) {
      sheet.style.backgroundImage = 'none';
      showToastNotification('Watermark configured for export', 'verified');
    } else {
      showToastNotification('Watermark suppressed on preview', 'block');
    }
  }

  // Universal Toast Notification Creator (Injects into App Shell's container)
  function showToastNotification(message, iconName) {
    const container = document.getElementById('toastContainer') || document.body;
    const toast = document.createElement('div');
    toast.className = 'p-3 rounded-2xl bg-[#18181A] text-white text-xs shadow-2xl flex items-center gap-2.5 transition-all duration-300 pointer-events-auto border border-zinc-700/60 max-w-sm';
    toast.innerHTML = '<span class="material-symbols-outlined text-[#F5A3BE] text-base flex-shrink-0">' + (iconName || 'info') + '</span><span class="font-medium">' + message + '</span>';
    container.appendChild(toast);
    setTimeout(() => {
      toast.style.opacity = '0';
      toast.style.transform = 'translateY(6px)';
      setTimeout(() => toast.remove(), 300);
    }, 3200);
  }

  // State flag to track whether user manually toggled calendar
  let userManuallyToggled = false;
  let isCalendarContracted = false;

  function updateCalendarButtonState(contracted) {
    const toggleText = document.getElementById('toggleCalendarText');
    const toggleBtn = document.getElementById('hide-calendar-btn');
    const toggleIcon = document.getElementById('toggleCalendarIcon');

    if (contracted) {
      if (toggleText) toggleText.textContent = 'Show Calendar';
      if (toggleIcon) toggleIcon.textContent = 'calendar_month';
      if (toggleBtn) {
        toggleBtn.setAttribute('title', 'Expand Calendar Panel');
        toggleBtn.classList.remove('bg-white', 'text-zinc-800');
        toggleBtn.classList.add('bg-zinc-900', 'text-white', 'ring-2', 'ring-[#F5A3BE]/40');
      }
    } else {
      if (toggleText) toggleText.textContent = 'Hide Calendar';
      if (toggleIcon) toggleIcon.textContent = 'view_sidebar';
      if (toggleBtn) {
        toggleBtn.setAttribute('title', 'Hide Calendar Panel');
        toggleBtn.classList.remove('bg-zinc-900', 'text-white', 'ring-2', 'ring-[#F5A3BE]/40');
        toggleBtn.classList.add('bg-white', 'text-zinc-800');
      }
    }
  }

  // Toggle Calendar Functionality
  function toggleCalendar(forceState) {
    const sidebar = document.getElementById('calendar-sidebar');
    if (!sidebar) return;

    if (typeof forceState === 'boolean') {
      isCalendarContracted = forceState;
    } else {
      userManuallyToggled = true; // explicitly clicked by user
      isCalendarContracted = !isCalendarContracted;
    }

    if (isCalendarContracted) {
      sidebar.classList.add('contracted-sidebar');
    } else {
      sidebar.classList.remove('contracted-sidebar');
    }

    updateCalendarButtonState(isCalendarContracted);
  }

  // Expose toggleCalendar globally
  window.toggleCalendar = toggleCalendar;
  window.toggleCalendarSidebar = toggleCalendar;

  // Auto-contraction on scroll logic
  function setupScrollAutoContract() {
    const scrollContainer = document.getElementById('mainWorkspace') || window;
    const scrollThreshold = 140; // threshold past top hero/header

    const handleScroll = () => {
      // Determine vertical scroll distance
      const scrollTop = scrollContainer === window 
        ? (window.pageYOffset || document.documentElement.scrollTop) 
        : scrollContainer.scrollTop;

      // If user hasn't explicitly locked state by manual click, or when they return to top
      if (scrollTop > scrollThreshold) {
        if (!isCalendarContracted && !userManuallyToggled) {
          toggleCalendar(true); // auto contract smoothly
        }
      } else if (scrollTop <= 40) {
        // When scrolled back to the very top, auto-expand back and reset manual override lock
        if (isCalendarContracted) {
          userManuallyToggled = false;
          toggleCalendar(false); // auto restore
        }
      }
    };

    if (scrollContainer === window) {
      window.addEventListener('scroll', handleScroll, { passive: true });
    } else {
      scrollContainer.addEventListener('scroll', handleScroll, { passive: true });
      window.addEventListener('scroll', handleScroll, { passive: true });
    }
  }

  __sfReady( () => {
    setupScrollAutoContract();

    // Bento card expansion handling if present
    const cards = document.querySelectorAll('.bento-card');
    let activeCardId = null;

    const initialExpanded = document.querySelector('.bento-card.is-expanded');
    if (initialExpanded) {
      activeCardId = initialExpanded.getAttribute('data-card-id');
    }

    function resetCards() {
      activeCardId = null;
      cards.forEach(card => {
        card.classList.remove('is-expanded', 'is-dimmed', 'ring-2', 'ring-zinc-900/30', 'shadow-xl');
        const defaultCol = card.getAttribute('data-default-col');
        if (defaultCol) {
          card.className = card.className.replace(/lg:col-span-\d+/g, '').trim();
          card.classList.add(defaultCol);
        }
        const badge = card.querySelector('.card-badge');
        if (badge) badge.classList.add('hidden');
        const path = card.querySelector('.expand-icon-path');
        if (path) {
          path.setAttribute('d', 'M15 3h6v6M9 21H3v-6M21 3l-7 7M3 21l7-7');
        }
      });
    }

    function expandCard(card) {
      const cardId = card.getAttribute('data-card-id');
      if (activeCardId === cardId) {
        resetCards();
        return;
      }
      activeCardId = cardId;
      cards.forEach(c => {
        const cId = c.getAttribute('data-card-id');
        const defaultCol = c.getAttribute('data-default-col');
        if (defaultCol) {
          c.className = c.className.replace(/lg:col-span-\d+/g, '').trim();
        }
        const badge = c.querySelector('.card-badge');
        const path = c.querySelector('.expand-icon-path');

        if (cId === cardId) {
          c.classList.remove('is-dimmed');
          c.classList.add('is-expanded', 'ring-2', 'ring-zinc-900/30', 'shadow-xl');
          c.classList.add('lg:col-span-12');
          if (badge) badge.classList.remove('hidden');
          if (path) path.setAttribute('d', 'M18 6L6 18M6 6l12 12');
        } else {
          c.classList.remove('is-expanded', 'ring-2', 'ring-zinc-900/30', 'shadow-xl');
          c.classList.add('is-dimmed');
          c.classList.add('lg:col-span-4');
          if (badge) badge.classList.add('hidden');
          if (path) path.setAttribute('d', 'M15 3h6v6M9 21H3v-6M21 3l-7 7M3 21l7-7');
        }
      });
    }

    cards.forEach(card => {
      card.addEventListener('click', (e) => {
        if (e.target.closest('button:not(.toggle-btn)') || e.target.closest('a')) return;
        expandCard(card);
      });
    });

    document.addEventListener('click', (e) => {
      if (!e.target.closest('#bentoGrid')) {
        if (activeCardId !== null) resetCards();
      }
    });

    document.addEventListener('keydown', (e) => {
      if (e.key === 'Escape' && activeCardId !== null) resetCards();
    });

    // Modal Handlers
    const aiModal = document.getElementById('aiSettingsModal');
    const aiTrigger = document.getElementById('aiSettingsTriggerBtn');
    const aiClose = document.getElementById('closeSettingsBtn');
    const saveSettingsBtn = document.getElementById('saveSettingsBtn');

    if (aiTrigger && aiModal) {
      aiTrigger.addEventListener('click', () => {
        aiModal.classList.remove('hidden');
        aiModal.classList.add('flex');
      });
    }
    if (aiClose && aiModal) {
      aiClose.addEventListener('click', () => {
        aiModal.classList.add('hidden');
        aiModal.classList.remove('flex');
      });
    }
    if (saveSettingsBtn && aiModal) {
      saveSettingsBtn.addEventListener('click', () => {
        aiModal.classList.add('hidden');
        aiModal.classList.remove('flex');
      });
    }

    const eventModal = document.getElementById('addEventModal');
    const closeEventBtn = document.getElementById('closeAddEventBtn');
    const cancelEventBtn = document.getElementById('cancelAddEventBtn');
    const confirmAddEventBtn = document.getElementById('confirmAddEventBtn');

    [closeEventBtn, cancelEventBtn].forEach(btn => {
      if (btn && eventModal) {
        btn.addEventListener('click', () => {
          eventModal.classList.add('hidden');
          eventModal.classList.remove('flex');
        });
      }
    });

    if (confirmAddEventBtn && eventModal) {
      confirmAddEventBtn.addEventListener('click', () => {
        eventModal.classList.add('hidden');
        eventModal.classList.remove('flex');
        const toast = document.getElementById('toastContainer');
        if (toast) {
          const item = document.createElement('div');
          item.className = 'p-3 rounded-2xl bg-[#18181A] text-white text-xs shadow-xl flex items-center gap-2';
          item.innerHTML = '<span class="material-symbols-outlined text-[#F5A3BE] text-base">event_available</span><span>Milestone registered on calendar!</span>';
          toast.appendChild(item);
          setTimeout(() => item.remove(), 3500);
        }
      });
    }
  });
