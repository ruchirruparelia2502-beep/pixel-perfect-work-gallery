  let activeCellBtn = null;
  let activeSelectedValue = 0;

  // Setup Cell Event Listeners
  __sfReady( () => {
    setupCellListeners();
  });

  function setupCellListeners() {
    const cells = document.querySelectorAll('.matrix-cell');
    cells.forEach(cell => {
      cell.addEventListener('click', (e) => {
        e.stopPropagation();
        openInspector(cell);
      });
    });
  }

  function openInspector(cellElement) {
    activeCellBtn = cellElement;
    const co = cellElement.getAttribute('data-co');
    const outcome = cellElement.getAttribute('data-outcome');
    const val = parseInt(cellElement.getAttribute('data-val') || '0', 10);
    const warning = cellElement.getAttribute('data-warning');

    activeSelectedValue = val;

    // Populate Inspector UI
    document.getElementById('inspectorTitle').innerText = `${co} × ${outcome} Correlation`;
    document.getElementById('inspectorCoTag').innerText = `${co} Overview`;
    document.getElementById('inspectorBloomTag').innerText = coKnowledge[co]?.bloom || 'L3 Apply';
    document.getElementById('inspectorCoStatement').innerText = coKnowledge[co]?.statement || '';
    document.getElementById('inspectorPoTag').innerText = `Target: ${outcome}`;
    document.getElementById('inspectorPoStatement').innerText = outcomeKnowledge[outcome] || `${outcome}: Outcome descriptor for distributed architecture curriculum.`;

    // Warning Banner logic
    const warnBox = document.getElementById('inspectorWarningBox');
    if (warning) {
      warnBox.classList.remove('hidden');
      document.getElementById('inspectorWarningTitle').innerText = `Senate Audit Flag (${outcome})`;
      document.getElementById('inspectorWarningDesc').innerText = warning;
    } else {
      warnBox.classList.add('hidden');
    }

    // Default Justification text
    const justificationArea = document.getElementById('justificationText');
    if (val === 3) {
      justificationArea.value = `Substantial correlation: Laboratory modules 3 & 4 provide direct hands-on implementation and evaluated rubric checkpoints for ${co} directly meeting ${outcome}.`;
    } else if (val === 2) {
      justificationArea.value = `Moderate correlation: Addressed in Module 2 lecture notes and tested via Continuous Internal Evaluation (CIE) Quiz 2 and midterm questions.`;
    } else if (val === 1) {
      justificationArea.value = `Slight correlation: Theoretical context provided during initial survey sessions without isolated lab rubric scoring.`;
    } else {
      justificationArea.value = `No direct alignment planned under R22 distributed systems syllabus directives.`;
    }

    updateStrengthButtonsUI(val);

    // Show slide-over
    document.getElementById('inspectorBackdrop').classList.remove('hidden');
    const panel = document.getElementById('mappingInspector');
    panel.classList.remove('translate-x-full');
  }

  function closeInspector() {
    const panel = document.getElementById('mappingInspector');
    panel.classList.add('translate-x-full');
    document.getElementById('inspectorBackdrop').classList.add('hidden');
  }

  function setStrength(strength) {
    activeSelectedValue = strength;
    updateStrengthButtonsUI(strength);
  }

  function updateStrengthButtonsUI(val) {
    for (let i = 0; i <= 3; i++) {
      const btn = document.getElementById(`strengthBtn${i}`);
      if (!btn) continue;
      if (i === val) {
        btn.classList.add('bg-[#18181A]', 'text-white', 'shadow-md');
        btn.classList.remove('bg-white', 'text-zinc-800', 'text-zinc-600');
      } else {
        btn.classList.remove('bg-[#18181A]', 'text-white', 'shadow-md');
        btn.classList.add('bg-white', 'text-zinc-800');
      }
    }
  }

  function saveInspectorData() {
    if (!activeCellBtn) return;

    activeCellBtn.setAttribute('data-val', activeSelectedValue);
    activeCellBtn.innerText = activeSelectedValue > 0 ? activeSelectedValue : '-';

    // Update styling based on strength
    activeCellBtn.classList.remove('bg-[#18181A]', 'bg-[#4B5563]', 'bg-[#E5DECD]', 'text-white', 'text-zinc-900', 'text-zinc-400');

    if (activeSelectedValue === 3) {
      activeCellBtn.classList.add('bg-[#18181A]', 'text-white', 'font-bold');
    } else if (activeSelectedValue === 2) {
      activeCellBtn.classList.add('bg-[#4B5563]', 'text-white', 'font-bold');
    } else if (activeSelectedValue === 1) {
      activeCellBtn.classList.add('bg-[#E5DECD]', 'text-zinc-900', 'font-bold');
    } else {
      activeCellBtn.classList.add('text-zinc-400');
    }

    // Recalculate row average for this CO
    const co = activeCellBtn.getAttribute('data-co');
    recalculateRowMean(co);

    closeInspector();
    showToast(`Updated ${co} × ${activeCellBtn.getAttribute('data-outcome')} to level ${activeSelectedValue}`);
  }

  function recalculateRowMean(coId) {
    const rowCells = document.querySelectorAll(`.matrix-cell[data-co="${coId}"]`);
    let sum = 0;
    let count = 0;
    rowCells.forEach(c => {
      const v = parseInt(c.getAttribute('data-val') || '0', 10);
      if (v > 0) {
        sum += v;
        count++;
      }
    });

    const mean = count > 0 ? (sum / count).toFixed(2) : '0.00';
    const meanDisplay = document.getElementById(`mean-${coId}`);
    if (meanDisplay) {
      meanDisplay.innerText = mean;
    }
  }

  function revertMapping() {
    if (!activeCellBtn) return;
    const oldVal = parseInt(activeCellBtn.getAttribute('data-val') || '0', 10);
    setStrength(oldVal);
    showToast('Reverted to current saved state');
  }

  function showToast(msg) {
    const toast = document.getElementById('toastNotification');
    const msgEl = document.getElementById('toastMessage');
    if (!toast || !msgEl) return;
    msgEl.innerText = msg;
    toast.classList.remove('translate-y-[-100px]', 'opacity-0');
    toast.classList.add('translate-y-0', 'opacity-100');
    setTimeout(() => {
      toast.classList.add('translate-y-[-100px]', 'opacity-0');
      toast.classList.remove('translate-y-0', 'opacity-100');
    }, 2800);
  }

  let userManuallyToggled = false;
  let isContracted = false;

  // Toggle Calendar Functionality
  function toggleCalendar(manual = false) {
    const sidebar = document.getElementById('calendar-sidebar');
    const toggleText = document.getElementById('toggleCalendarText');
    const toggleBtn = document.getElementById('hide-calendar-btn');

    if (!sidebar) return;

    if (manual) {
      userManuallyToggled = true;
    }

    if (isContracted) {
      // Reopen sidebar
      sidebar.classList.remove('contracted');
      isContracted = false;
      if (toggleText) toggleText.textContent = 'Hide Calendar';
      if (toggleBtn) {
        toggleBtn.setAttribute('title', 'Hide Calendar Panel');
        toggleBtn.classList.remove('bg-zinc-900', 'text-[#F5A3BE]');
        toggleBtn.classList.add('bg-[#F6F1E5]', 'text-zinc-800');
      }
    } else {
      // Collapse sidebar
      sidebar.classList.add('contracted');
      isContracted = true;
      if (toggleText) toggleText.textContent = 'Show Calendar';
      if (toggleBtn) {
        toggleBtn.setAttribute('title', 'Show Calendar Panel');
        toggleBtn.classList.remove('bg-[#F6F1E5]', 'text-zinc-800');
        toggleBtn.classList.add('bg-zinc-900', 'text-[#F5A3BE]');
      }
    }
  }

  // Auto-contraction on scroll
  function setupScrollAutoContract() {
    const workspace = document.getElementById('mainWorkspace');
    if (!workspace) return;

    const SCROLL_THRESHOLD = 140; // Past header / action toolbar

    workspace.addEventListener('scroll', () => {
      const currentScroll = workspace.scrollTop;

      // If user has not explicitly intervened or if scrolled significantly down
      if (!userManuallyToggled) {
        if (currentScroll > SCROLL_THRESHOLD && !isContracted) {
          toggleCalendar(false);
        } else if (currentScroll <= 30 && isContracted) {
          // Auto expand back when scrolling back near the top
          toggleCalendar(false);
        }
      } else {
        // If user scrolled all the way to the top, reset manual override latch
        if (currentScroll <= 10) {
          userManuallyToggled = false;
        }
      }
    }, { passive: true });
  }

  // Expose toggleCalendar globally
  window.toggleCalendar = toggleCalendar;
  window.toggleCalendarSidebar = toggleCalendar;

  __sfReady( () => {
    setupScrollAutoContract();

    const cards = document.querySelectorAll('.bento-card');
    let activeCardId = null;

    // Check if yellow card is initially expanded
    const initialExpanded = document.querySelector('.bento-card.is-expanded');
    if (initialExpanded) {
      activeCardId = initialExpanded.getAttribute('data-card-id');
    }

    function resetCards() {
      activeCardId = null;
      cards.forEach(card => {
        card.classList.remove('is-expanded', 'is-dimmed', 'ring-2', 'ring-zinc-900/30', 'shadow-xl');
        
        // Restore default column spans
        const defaultCol = card.getAttribute('data-default-col');
        card.className = card.className.replace(/lg:col-span-\d+/g, '').trim();
        card.classList.add(defaultCol);

        // Hide badges
        const badge = card.querySelector('.card-badge');
        if (badge) badge.classList.add('hidden');

        // Reset icon to expand icon
        const path = card.querySelector('.expand-icon-path');
        if (path) {
          path.setAttribute('d', 'M15 3h6v6M9 21H3v-6M21 3l-7 7M3 21l7-7');
        }
      });
    }

    function expandCard(card) {
      const cardId = card.getAttribute('data-card-id');

      // If already active, clicking again collapses it
      if (activeCardId === cardId) {
        resetCards();
        return;
      }

      activeCardId = cardId;

      cards.forEach(c => {
        const cId = c.getAttribute('data-card-id');
        const defaultCol = c.getAttribute('data-default-col');
        c.className = c.className.replace(/lg:col-span-\d+/g, '').trim();

        const badge = c.querySelector('.card-badge');
        const path = c.querySelector('.expand-icon-path');

        if (cId === cardId) {
          c.classList.remove('is-dimmed');
          c.classList.add('is-expanded', 'ring-2', 'ring-zinc-900/30', 'shadow-xl');
          c.classList.add('lg:col-span-12'); // Take full width prominence in grid
          if (badge) badge.classList.remove('hidden');
          if (path) {
            // Collapse cross icon '✕'
            path.setAttribute('d', 'M18 6L6 18M6 6l12 12');
          }
        } else {
          c.classList.remove('is-expanded', 'ring-2', 'ring-zinc-900/30', 'shadow-xl');
          c.classList.add('is-dimmed');
          // For inactive cards, distribute compactly
          c.classList.add('lg:col-span-4');
          if (badge) badge.classList.add('hidden');
          if (path) {
            path.setAttribute('d', 'M15 3h6v6M9 21H3v-6M21 3l-7 7M3 21l7-7');
          }
        }
      });
    }

    // Attach click listeners to cards and their toggle buttons
    cards.forEach(card => {
      card.addEventListener('click', (e) => {
        // Prevent click if clicking direct action links inside expanded area
        if (e.target.closest('button:not(.toggle-btn)') || e.target.closest('a')) {
          return;
        }
        expandCard(card);
      });
    });

    // Clicking outside the grid closes any open card
    document.addEventListener('click', (e) => {
      if (!e.target.closest('#bentoGrid')) {
        if (activeCardId !== null) {
          resetCards();
        }
      }
    });

    // Allow Esc key to contract cards
    document.addEventListener('keydown', (e) => {
      if (e.key === 'Escape' && activeCardId !== null) {
        resetCards();
      }
    });

    // AI Provider modal handlers
    const aiModal = document.getElementById('aiSettingsModal');
    const aiTrigger = document.getElementById('aiSettingsTriggerBtn');
    const aiClose = document.getElementById('closeSettingsBtn');
    const saveSettingsBtn = document.getElementById('saveSettingsBtn');

    if (aiTrigger && aiModal) {
      aiTrigger.addEventListener('click', () => {
        aiModal.classList.remove('hidden');
        aiModal.classList.add('flex');
      });
    }
    if (aiClose && aiModal) {
      aiClose.addEventListener('click', () => {
        aiModal.classList.add('hidden');
        aiModal.classList.remove('flex');
      });
    }
    if (saveSettingsBtn && aiModal) {
      saveSettingsBtn.addEventListener('click', () => {
        aiModal.classList.add('hidden');
        aiModal.classList.remove('flex');
      });
    }
  });
